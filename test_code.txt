3. 수정 (수정 폼 → confirm → 실제 API)
const handleEditClick = (row: InstitutionRow) => {
  open({
    type: "content",
    title: "기관 수정",
    content: (
      <FieldForm
        fields={institutionFieldConfig}
        defaultValues={row}
        onSubmit={async (data) => {
          open({
            type: "confirm",
            title: "수정하시겠습니까?",
            description: "변경사항을 저장합니다.",
            onConfirm: async () => {
              await updateMutation.mutateAsync({ id: row.id, data });
            },
          });
        }}
      />
    ),
  });
};
